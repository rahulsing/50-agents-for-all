    Agent 53 (extra) - Notes agent over MCP (LangGraph client)¶

    Framework: LangGraph + MCP  ·  Industry: Protocols  ·  Source folder on GitHub

    What it does¶

    The same notes assistant as agent 52, built with LangGraph instead of Agno, using the same MCP server (../notes_mcp_server.py) and the same notes.json. Notes saved by one agent are visible to the other.

    Concepts you learn¶

    • MCP makes tools framework-independent: no tool code was rewritten for LangGraph. MultiServerMCPClient (from langchain-mcp-adapters) connects to the server and converts its tools into LangChain tools.
    • Several servers at once: MultiServerMCPClient({...}) takes a dict, so one agent can combine tools from many MCP servers (notes, files, a database...).
    • Version independence: this venv has mcp 1.x and the Agno venv has mcp 2.x; both talk to the same server because MCP is a protocol, not a library API.

    The flow¶

    request -> create_agent (LangGraph) -> MCP tools -> notes_mcp_server.py -> notes.json

    Run¶

    cd 50-agents-for-all
.venv-langgraph\Scripts\python.exe protocols\53_notes_mcp_langgraph\main.py "What notes do I have about shopping?"
.venv-langgraph\Scripts\python.exe protocols\53_notes_mcp_langgraph\main.py "Delete the groceries note"
    Empty request exits with code 1. The first line printed is the list of tools discovered from the server.

    Things to try¶

    • Add a second server to the dict (any public MCP server) and ask questions that need both.
    • Compare the code size with agent 52.

    Known limits¶

    • Each run starts a fresh server process; fine for demos, slower than a long-running server.

    """LangGraph agent that uses the SAME MCP server as agent 52, via langchain-mcp-adapters."""
import asyncio
import sys
from pathlib import Path

sys.path.insert(0, str(Path(__file__).resolve().parents[2]))

from langchain.agents import create_agent
from langchain_mcp_adapters.client import MultiServerMCPClient
from langchain_ollama import ChatOllama

from common.llm import DEFAULT_MODEL, OLLAMA_URL

SERVER = Path(__file__).resolve().parents[1] / "notes_mcp_server.py"


async def run(request: str) -> None:
    client = MultiServerMCPClient({
        "notes": {"command": sys.executable, "args": [str(SERVER)], "transport": "stdio"},
    })
    tools = await client.get_tools()  # MCP tools converted into LangChain tools
    print(f"[mcp] tools from server: {[t.name for t in tools]}")
    agent = create_agent(ChatOllama(model=DEFAULT_MODEL, base_url=OLLAMA_URL, temperature=0), tools,
                         system_prompt="You manage the user's notes with the notes tools. Search or list notes "
                                       "before answering questions about them; never invent notes.")
    async for step in agent.astream({"messages": [("user", request)]}, stream_mode="values"):
        msg = step["messages"][-1]
        for call in getattr(msg, "tool_calls", None) or []:
            print(f"[tool call] {call['name']}({call['args']})")
    print(f"\n{msg.content}")


def main() -> None:
    request = " ".join(sys.argv[1:]).strip()
    if not request:
        print('Please give a request, e.g. main.py "What notes do I have about shopping?"')
        sys.exit(1)
    asyncio.run(run(request))


if __name__ == "__main__":
    main()

    """A tiny MCP server that manages notes in a JSON file. Any MCP client (Agno, LangGraph, Claude Code, ...) can use it.

Run directly to check it starts:  python notes_mcp_server.py   (it then waits for a client on stdin/stdout)
"""
import json
from datetime import datetime
from pathlib import Path

try:  # mcp 1.x
    from mcp.server.fastmcp import FastMCP as Server
except ImportError:  # mcp 2.x renamed FastMCP to MCPServer
    from mcp.server.mcpserver import MCPServer as Server

NOTES = Path(__file__).parent / "notes.json"
server = Server("notes")


def _load() -> list[dict]:
    return json.loads(NOTES.read_text(encoding="utf-8")) if NOTES.exists() else []


def _save(notes: list[dict]) -> None:
    NOTES.write_text(json.dumps(notes, indent=2), encoding="utf-8")


@server.tool()
def add_note(title: str, text: str, tags: list[str] | None = None) -> str:
    """Save a new note with a title, text and optional tags."""
    notes = _load()
    note = {"id": len(notes) + 1, "title": title, "text": text, "tags": tags or [],
            "created": datetime.now().isoformat(timespec="minutes")}
    notes.append(note)
    _save(notes)
    return f"Saved note {note['id']}: {title}"


@server.tool()
def list_notes() -> str:
    """List all notes (id, title, tags)."""
    notes = _load()
    return "\n".join(f"{n['id']}. {n['title']} {n['tags']}" for n in notes) or "No notes yet."


@server.tool()
def search_notes(query: str) -> str:
    """Find notes whose title, text or tags contain the query (case-insensitive)."""
    q = query.lower()
    hits = [n for n in _load() if q in n["title"].lower() or q in n["text"].lower()
            or any(q in t.lower() for t in n["tags"])]
    return "\n\n".join(f"{n['id']}. {n['title']}\n{n['text']}" for n in hits) or f"No notes match '{query}'."


@server.tool()
def delete_note(note_id: int) -> str:
    """Delete a note by its id."""
    notes = _load()
    kept = [n for n in notes if n["id"] != note_id]
    if len(kept) == len(notes):
        return f"No note with id {note_id}."
    _save(kept)
    return f"Deleted note {note_id}."


if __name__ == "__main__":
    server.run()  # default transport: stdio
